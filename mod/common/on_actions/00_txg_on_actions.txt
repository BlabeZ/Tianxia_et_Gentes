# 天下与万邦 全局政治状态钩子（T-049，D-20260813-002）
# 项目党派切换由 TXG_set_project_party 原子完成；原生换党只校验，不反推具体党派。

on_startup = {
	effect = {
		every_country = {
			TXG_validate_project_party_state = yes
			if = {
				limit = { has_country_flag = TXG_party_state_ready }
				set_country_flag = TXG_opinion_refresh_required
			}
		}
	}
}

on_ruling_party_change = {
	effect = {
		if = {
			limit = { NOT = { has_country_flag = TXG_party_sync_in_progress } }
			TXG_validate_project_party_state = yes
			if = {
				limit = { has_country_flag = TXG_party_state_ready }
				set_country_flag = TXG_opinion_refresh_required
			}
		}
	}
}
